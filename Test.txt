#Requires -Version 5.1
<#
.SYNOPSIS
    Keeps a SharePoint Online document set under 4500 items and moves older overflow to a target document set.

.DESCRIPTION
    Reads a CSV file containing source and target SharePoint Online site URLs and document set paths.
    For each row, the script connects to the source site, inspects the document set, and if the item
    count exceeds the configured cap (default 4500), the oldest items are moved to the target document
    set in the target site collection until the source document set is at or under the cap.

    This is intended for SharePoint Online document sets that hit the 5000-item threshold and need
    the older content split into a separate overflow document set while keeping the active document
    set under the recommended operating threshold.

.PARAMETER CsvInputPath
    Path to the CSV file with required columns:
    - SourceSiteUrl         : SharePoint Online source site URL, for example https://tenant.sharepoint.com/sites/Finance
    - SourceDocumentSetPath : Source document set path, for example /sites/Finance/Shared Documents/DocSet-Active
    - TargetSiteUrl         : SharePoint Online target site URL, for example https://tenant.sharepoint.com/sites/Archive
    - TargetDocumentSetPath : Overflow document set path, for example /sites/Archive/Shared Documents/DocSet-Archive

    The script also accepts the legacy single-site format using SiteUrl + TargetDocumentSetPath when
    the source and target site collection are the same.

.PARAMETER ClientId
    Azure AD application client ID used for SharePoint Online authentication.

.PARAMETER TargetUsername
    SharePoint Online username used for authentication.

.PARAMETER TargetPassword
    SharePoint Online password used for authentication.

.PARAMETER MaxAllowedItems
    Maximum number of items to keep in the active source document set.
    Default: 4500

.PARAMETER OutputFolder
    Folder path for output files.
    Default: ./SplitDocumentSetLog-{timestamp}

.EXAMPLE
    .\Split-DocumentSetByItemCount.ps1 `
        -CsvInputPath ".\DocumentSetSplit-Input.csv" `
        -ClientId "xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx" `
        -TargetUsername "admin@contoso.onmicrosoft.com" `
        -TargetPassword "Password123!" `
        -MaxAllowedItems 4500 `
        -OutputFolder ".\SplitDocumentSetLog"

    .\Split-DocumentSetByItemCount.ps1 `
        -CsvInputPath ".\DocumentSetSplit-Input.csv" `
        -ClientId "xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx" `
        -TargetUsername "admin@contoso.onmicrosoft.com" `
        -TargetPassword "Password123!" `
        -MaxAllowedItems 4500 `
        -PreviewOnly `
        -OutputFolder ".\SplitDocumentSetPreview"

    .\Split-DocumentSetByItemCount.ps1 `
        -CsvInputPath ".\DocumentSetSplit-Input.csv" `
        -ClientId "xxxxxxxx-xxxx-xxxx-xxxx-xxxxxxxxxxxx" `
        -TargetUsername "admin@contoso.onmicrosoft.com" `
        -TargetPassword "Password123!" `
        -MaxAllowedItems 4500 `
        -PreviewOnly `
        -OutputFolder ".\SplitDocumentSetPreview"
#>

[CmdletBinding()]
param(
    [Parameter(Mandatory = $true)]
    [string]$CsvInputPath,

    [Parameter(Mandatory = $true)]
    [string]$ClientId,

    [Parameter(Mandatory = $true)]
    [string]$TargetUsername,

    [Parameter(Mandatory = $true)]
    [string]$TargetPassword,

    [int]$MaxAllowedItems = 4500,

    [switch]$PreviewOnly,

    [string]$OutputFolder = "./SplitDocumentSetLog-$(Get-Date -Format 'yyyyMMdd-HHmmss')"
)

Set-StrictMode -Version Latest
$ErrorActionPreference = "Continue"

# ==========================================
# Global Variables & Initialization
# ==========================================
$script:ProcessedRows  = 0
$script:SuccessfulRows = 0
$script:FailedRows     = 0
$script:Credential     = $null
$script:CurrentSiteUrl = ""

$script:ResultWriter   = $null
$script:ResultFilePath = $null
$script:PreviewWriter  = $null
$script:PreviewFilePath = $null
$script:ResultColumns  = @(
    "RowNumber",
    "SourceSiteUrl",
    "TargetSiteUrl",
    "SourceDocumentSetPath",
    "TargetDocumentSetPath",
    "SourceItemCount",
    "TargetItemCount",
    "MovedItemCount",
    "RetainedItemCount",
    "Status",
    "IsSuccessful",
    "Message",
    "Timestamp"
)

if (-not (Test-Path $CsvInputPath)) {
    throw "CSV input file not found: $CsvInputPath"
}

if (-not (Test-Path $OutputFolder)) {
    New-Item -ItemType Directory -Path $OutputFolder -Force | Out-Null
    Write-Host "Created output folder: $OutputFolder"
}

# ==========================================
# Function: Initialize Credential
# ==========================================
function Initialize-Credential {
    [CmdletBinding()]
    param()

    $securePassword = ConvertTo-SecureString $TargetPassword -AsPlainText -Force
    $script:Credential = New-Object System.Management.Automation.PSCredential($TargetUsername, $securePassword)
}

# ==========================================
# Function: Connect to SPO Site
# ==========================================
function Get-PnPConnectionForSite {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [string]$SiteUrl
    )

    if ($script:CurrentSiteUrl -eq $SiteUrl) {
        return
    }

    try {
        Write-Verbose "Connecting to $SiteUrl"
        Connect-PnPOnline -Url $SiteUrl -Credentials $script:Credential -ClientId $ClientId -ErrorAction Stop
        $script:CurrentSiteUrl = $SiteUrl
        Write-Host "Connected: $SiteUrl"
    }
    catch {
        throw "Failed to connect to site $SiteUrl : $_"
    }
}

# ==========================================
# Function: Validate CSV Columns
# ==========================================
function Test-CsvFormat {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [array]$Rows
    )

    if ($Rows.Count -eq 0) {
        throw "CSV file is empty: $CsvInputPath"
    }

    $first = $Rows[0]
    $props = $first.PSObject.Properties.Name

    if (($props -contains "SourceSiteUrl") -and ($props -contains "SourceDocumentSetPath") -and ($props -contains "TargetSiteUrl") -and ($props -contains "TargetDocumentSetPath")) {
        return
    }

    if (($props -contains "SiteUrl") -and ($props -contains "SourceDocumentSetPath") -and ($props -contains "TargetDocumentSetPath")) {
        return
    }

    throw "CSV must contain either SourceSiteUrl/TargetSiteUrl + SourceDocumentSetPath/TargetDocumentSetPath or the legacy SiteUrl + SourceDocumentSetPath + TargetDocumentSetPath columns."
}

# ==========================================
# Function: Initialize Result Writer
# ==========================================
function Initialize-ResultWriter {
    [CmdletBinding()]
    param()

    $timestamp = Get-Date -Format 'yyyyMMdd-HHmmss'
    $script:ResultFilePath = Join-Path $OutputFolder "Results_$timestamp.csv"

    $encoding = New-Object System.Text.UTF8Encoding($false)
    $script:ResultWriter = New-Object System.IO.StreamWriter($script:ResultFilePath, $false, $encoding)
    $script:ResultWriter.AutoFlush = $true
    $script:ResultWriter.WriteLine((ConvertTo-CsvLine -Values $script:ResultColumns))
}

# ==========================================
# Function: Write One Result Row
# ==========================================
function Write-ResultRow {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [PSCustomObject]$Row
    )

    $values = foreach ($col in $script:ResultColumns) {
        if ($Row.PSObject.Properties.Name -contains $col) { $Row.$col } else { "" }
    }
    $script:ResultWriter.WriteLine((ConvertTo-CsvLine -Values $values))
}

# ==========================================
# Function: Initialize Preview Writer
# ==========================================
function Initialize-PreviewWriter {
    [CmdletBinding()]
    param()

    $timestamp = Get-Date -Format 'yyyyMMdd-HHmmss'
    $script:PreviewFilePath = Join-Path $OutputFolder "DocumentSetMovePlan_$timestamp.csv"

    $encoding = New-Object System.Text.UTF8Encoding($false)
    $script:PreviewWriter = New-Object System.IO.StreamWriter($script:PreviewFilePath, $false, $encoding)
    $script:PreviewWriter.AutoFlush = $true

    $previewColumns = @(
        "SiteUrl",
        "SourceDocumentSetPath",
        "TargetDocumentSetPath",
        "FileName",
        "FileUrl",
        "Action",
        "ModifiedDate",
        "MoveReason",
        "Timestamp"
    )

    $script:PreviewWriter.WriteLine((ConvertTo-CsvLine -Values $previewColumns))
}

# ==========================================
# Function: CSV-Quote One Line
# ==========================================
function ConvertTo-CsvLine {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [object[]]$Values
    )

    $escaped = foreach ($v in $Values) {
        if ($null -eq $v) {
            ''
        }
        else {
            $s = [string]$v
            if ($s -match '[,"\r\n]') {
                '"' + ($s -replace '"', '""') + '"'
            }
            else {
                $s
            }
        }
    }
    [string]::Join(',', $escaped)
}

# ==========================================
# Function: Resolve Document Set URL
# ==========================================
function Resolve-DocumentSetUrl {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [string]$SiteUrl,

        [Parameter(Mandatory = $true)]
        [string]$DocumentSetPath
    )

    $trimmedPath = $DocumentSetPath.Trim()
    if ([string]::IsNullOrWhiteSpace($trimmedPath)) {
        throw "Document set path is empty."
    }

    if ($trimmedPath -match '^https?://') {
        return $trimmedPath.TrimEnd('/')
    }

    $base = $SiteUrl.TrimEnd('/')
    if ($trimmedPath.StartsWith('/')) {
        return "$base$trimmedPath"
    }

    return "$base/$trimmedPath"
}

# ==========================================
# Function: Ensure Document Set Exists
# ==========================================
function Ensure-DocumentSetExists {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [string]$SiteUrl,

        [Parameter(Mandatory = $true)]
        [string]$DocumentSetPath
    )

    $fullUrl = Resolve-DocumentSetUrl -SiteUrl $SiteUrl -DocumentSetPath $DocumentSetPath

    $existingFolder = Get-PnPFolder -Url $fullUrl -ErrorAction SilentlyContinue
    if ($null -ne $existingFolder) {
        return $fullUrl
    }

    $folderName = [System.IO.Path]::GetFileName($fullUrl.TrimEnd('/'))
    $parentUrl = $fullUrl.Substring(0, $fullUrl.LastIndexOf('/'))
    if ([string]::IsNullOrWhiteSpace($parentUrl)) {
        throw "Cannot determine parent folder for $DocumentSetPath"
    }

    $parentFolder = Get-PnPFolder -Url $parentUrl -ErrorAction Stop
    $folder = New-PnPFolder -Name $folderName -Folder $parentUrl -ErrorAction Stop
    return $folder.ServerRelativeUrl
}

# ==========================================
# Function: Get Files in Document Set
# ==========================================
function Get-DocumentSetFiles {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [string]$SiteUrl,

        [Parameter(Mandatory = $true)]
        [string]$DocumentSetPath
    )

    $fullUrl = Resolve-DocumentSetUrl -SiteUrl $SiteUrl -DocumentSetPath $DocumentSetPath
    $folder = Get-PnPFolder -Url $fullUrl -Includes Files -ErrorAction Stop

    if ($null -eq $folder.Files) {
        return @()
    }

    return @($folder.Files)
}

# ==========================================
# Function: Get Sort Key for File
# ==========================================
function Get-FileSortTimestamp {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [object]$File
    )

    $timeValue = $null

    if ($null -ne $File.TimeLastModified) {
        $timeValue = $File.TimeLastModified
    }
    elseif ($null -ne $File.TimeCreated) {
        $timeValue = $File.TimeCreated
    }
    elseif ($null -ne $File.LastModified) {
        $timeValue = $File.LastModified
    }
    elseif ($null -ne $File.Modified) {
        $timeValue = $File.Modified
    }

    if ($null -eq $timeValue) {
        return [DateTime]::MinValue
    }

    $dateValue = [DateTime]$timeValue
    return $dateValue
}

# ==========================================
# Function: Move Overflow Items to Target Document Set
# ==========================================
function Move-OverflowItemsToTargetDocumentSet {
    [CmdletBinding()]
    param(
        [Parameter(Mandatory = $true)]
        [string]$SiteUrl,

        [Parameter(Mandatory = $false)]
        [string]$TargetSiteUrl,

        [Parameter(Mandatory = $true)]
        [string]$SourceDocumentSetPath,

        [Parameter(Mandatory = $true)]
        [string]$TargetDocumentSetPath,

        [Parameter(Mandatory = $true)]
        [int]$MaxAllowedItems,

        [switch]$PreviewOnly
    )

    if ([string]::IsNullOrWhiteSpace($TargetSiteUrl)) {
        $TargetSiteUrl = $SiteUrl
    }

    $sourceFullUrl = Resolve-DocumentSetUrl -SiteUrl $SiteUrl -DocumentSetPath $SourceDocumentSetPath
    $targetFullUrl = Resolve-DocumentSetUrl -SiteUrl $TargetSiteUrl -DocumentSetPath $TargetDocumentSetPath

    if ($sourceFullUrl -eq $targetFullUrl) {
        throw "Source and target document set paths are identical. Please choose different paths."
    }

    $sourceFiles = @(Get-DocumentSetFiles -SiteUrl $SiteUrl -DocumentSetPath $SourceDocumentSetPath)
    $sourceItemCount = $sourceFiles.Count

    if ($sourceItemCount -le $MaxAllowedItems) {
        $targetFiles = @()
        if (-not $PreviewOnly) {
            $targetFullUrl = Ensure-DocumentSetExists -SiteUrl $TargetSiteUrl -DocumentSetPath $TargetDocumentSetPath
            $targetFiles = @(Get-DocumentSetFiles -SiteUrl $TargetSiteUrl -DocumentSetPath $TargetDocumentSetPath)
        }

        if ($PreviewOnly) {
            foreach ($file in $sourceFiles) {
                $previewRow = [PSCustomObject]@{
                    SourceSiteUrl = $SiteUrl
                    TargetSiteUrl = $TargetSiteUrl
                    SourceDocumentSetPath = $SourceDocumentSetPath
                    TargetDocumentSetPath = $TargetDocumentSetPath
                    FileName = $file.Name
                    FileUrl = $file.ServerRelativeUrl
                    Action = "Keep"
                    ModifiedDate = if ($null -ne $file.TimeLastModified) { $file.TimeLastModified.ToString("s") } else { "" }
                    MoveReason = "Within max item cap"
                    Timestamp = (Get-Date).ToString("s")
                }
                Write-PreviewRow -Row $previewRow
            }
        }

        return [PSCustomObject]@{
            SourceItemCount = $sourceItemCount
            TargetItemCount = $targetFiles.Count
            MovedItemCount = 0
            RetainedItemCount = $sourceItemCount
            Status = if ($PreviewOnly) { "PreviewOnlyNoAction" } else { "NoAction" }
            IsSuccessful = $true
            Message = "Source document set has $sourceItemCount items and is already within the $MaxAllowedItems item cap."
        }
    }

    if (-not $PreviewOnly) {
        $targetFullUrl = Ensure-DocumentSetExists -SiteUrl $TargetSiteUrl -DocumentSetPath $TargetDocumentSetPath
    }

    $overflowCount = $sourceItemCount - $MaxAllowedItems
    $sortedFiles = @($sourceFiles | Sort-Object { Get-FileSortTimestamp -File $_ })
    $filesToMove = @($sortedFiles | Select-Object -First $overflowCount)
    $moveLookup = @{}
    foreach ($file in $filesToMove) {
        if ($null -ne $file.ServerRelativeUrl) {
            $moveLookup[$file.ServerRelativeUrl] = $true
        }
    }
    $movedCount = 0

    if ($PreviewOnly) {
        foreach ($file in $sortedFiles) {
            $shouldMove = $moveLookup.ContainsKey($file.ServerRelativeUrl)
            $previewRow = [PSCustomObject]@{
                SourceSiteUrl = $SiteUrl
                TargetSiteUrl = $TargetSiteUrl
                SourceDocumentSetPath = $SourceDocumentSetPath
                TargetDocumentSetPath = $TargetDocumentSetPath
                FileName = $file.Name
                FileUrl = $file.ServerRelativeUrl
                Action = if ($shouldMove) { "Move" } else { "Keep" }
                ModifiedDate = if ($null -ne $file.TimeLastModified) { $file.TimeLastModified.ToString("s") } else { "" }
                MoveReason = if ($shouldMove) { "Older than max item threshold" } else { "Newest items kept in source" }
                Timestamp = (Get-Date).ToString("s")
            }
            Write-PreviewRow -Row $previewRow
            if ($shouldMove) { $movedCount++ }
        }

        $updatedTargetFiles = @()
        try {
            $updatedTargetFiles = @(Get-DocumentSetFiles -SiteUrl $TargetSiteUrl -DocumentSetPath $TargetDocumentSetPath)
        }
        catch {
            $updatedTargetFiles = @()
        }

        return [PSCustomObject]@{
            SourceItemCount = $sourceItemCount
            TargetItemCount = $updatedTargetFiles.Count
            MovedItemCount = $movedCount
            RetainedItemCount = $sourceItemCount - $movedCount
            Status = "PreviewOnlySplit"
            IsSuccessful = $true
            Message = "Preview plan: $movedCount oldest items would move from '$SourceDocumentSetPath' to '$TargetDocumentSetPath'. Source document set would retain $($sourceItemCount - $movedCount) items."
        }
    }

    foreach ($file in $filesToMove) {
        $sourceUrl = $file.ServerRelativeUrl
        $fileName = $file.Name
        $targetFileUrl = "$targetFullUrl/$fileName"

        try {
            Move-PnPFile -SourceUrl $sourceUrl -TargetUrl $targetFileUrl -Force -ErrorAction Stop
            $movedCount++
        }
        catch {
            throw "Failed to move file '$fileName' from '$sourceUrl' to '$targetFileUrl': $_"
        }
    }

    $updatedSourceFiles = @(Get-DocumentSetFiles -SiteUrl $SiteUrl -DocumentSetPath $SourceDocumentSetPath)
    $updatedTargetFiles = @(Get-DocumentSetFiles -SiteUrl $TargetSiteUrl -DocumentSetPath $TargetDocumentSetPath)
    $finalRetained = $updatedSourceFiles.Count

    return [PSCustomObject]@{
        SourceItemCount = $sourceItemCount
        TargetItemCount = $updatedTargetFiles.Count
        MovedItemCount = $movedCount
        RetainedItemCount = $finalRetained
        Status = "Split"
        IsSuccessful = $true
        Message = "Moved $movedCount oldest items from '$SourceDocumentSetPath' to '$TargetDocumentSetPath'. Source document set now contains $finalRetained items."
    }
}

# ==========================================
# Function: Invoke-SplitDocumentSetByItemCount
# ==========================================
function Invoke-SplitDocumentSetByItemCount {
    [CmdletBinding()]
    param()

    $startTime = Get-Date
    Write-Host "Reading CSV file: $CsvInputPath" -ForegroundColor Cyan

    if ($PreviewOnly) {
        Initialize-PreviewWriter
        Write-Host "Preview mode enabled. No files will be moved; a move plan CSV will be generated instead." -ForegroundColor Yellow
    }

    $rows = Import-Csv -Path $CsvInputPath
    Test-CsvFormat -Rows $rows

    $totalRows = $rows.Count
    Write-Host "Found $totalRows rows to process" -ForegroundColor Cyan

    $rowIndex = 0
    foreach ($csvRow in $rows) {
        $rowIndex++

        $siteUrl = if ($csvRow.PSObject.Properties.Name -contains "SourceSiteUrl") { $csvRow.SourceSiteUrl } else { $csvRow.SiteUrl }
        $targetSiteUrl = if ($csvRow.PSObject.Properties.Name -contains "TargetSiteUrl") { $csvRow.TargetSiteUrl } else { $siteUrl }
        $sourceDocSetPath = $csvRow.SourceDocumentSetPath
        $targetDocSetPath = $csvRow.TargetDocumentSetPath

        if ($rowIndex % 10 -eq 0) {
            Write-Host "Processing row $rowIndex of $totalRows..." -ForegroundColor Gray
        }

        # Skip blank rows.
        if ([string]::IsNullOrWhiteSpace($siteUrl) -or [string]::IsNullOrWhiteSpace($targetSiteUrl) -or [string]::IsNullOrWhiteSpace($sourceDocSetPath) -or [string]::IsNullOrWhiteSpace($targetDocSetPath)) {
            $row = [PSCustomObject]@{
                RowNumber = $rowIndex
                SourceSiteUrl = $siteUrl
                TargetSiteUrl = $targetSiteUrl
                SourceDocumentSetPath = $sourceDocSetPath
                TargetDocumentSetPath = $targetDocSetPath
                SourceItemCount = ""
                TargetItemCount = ""
                MovedItemCount = ""
                RetainedItemCount = ""
                Status = "Skipped"
                IsSuccessful = $false
                Message = "Missing required field (SourceSiteUrl/TargetSiteUrl, SourceDocumentSetPath, or TargetDocumentSetPath)"
                Timestamp = (Get-Date).ToString("s")
            }
            Write-ResultRow -Row $row
            $script:ProcessedRows++
            $script:FailedRows++
            Write-Warning "Row $rowIndex skipped: Missing required fields"
            continue
        }

        $connectionError = $null
        try {
            Get-PnPConnectionForSite -SiteUrl $siteUrl
        }
        catch {
            $connectionError = $_.Exception.Message
        }

        if ($connectionError) {
            $row = [PSCustomObject]@{
                RowNumber = $rowIndex
                SourceSiteUrl = $siteUrl
                TargetSiteUrl = $targetSiteUrl
                SourceDocumentSetPath = $sourceDocSetPath
                TargetDocumentSetPath = $targetDocSetPath
                SourceItemCount = ""
                TargetItemCount = ""
                MovedItemCount = ""
                RetainedItemCount = ""
                Status = "Failed"
                IsSuccessful = $false
                Message = "Connection failed: $connectionError"
                Timestamp = (Get-Date).ToString("s")
            }
            Write-ResultRow -Row $row
            $script:ProcessedRows++
            $script:FailedRows++
            Write-Warning "Row $rowIndex failed: Could not connect to $siteUrl"
            continue
        }

        $actionResult = $null
        $status = "Failed"
        $isSuccessful = $false
        $message = ""

        try {
            $actionResult = Move-OverflowItemsToTargetDocumentSet -SiteUrl $siteUrl -TargetSiteUrl $targetSiteUrl -SourceDocumentSetPath $sourceDocSetPath -TargetDocumentSetPath $targetDocSetPath -MaxAllowedItems $MaxAllowedItems -PreviewOnly:$PreviewOnly
            $status = $actionResult.Status
            $isSuccessful = $actionResult.IsSuccessful
            $message = $actionResult.Message
        }
        catch {
            $message = $_.Exception.Message
            $status = "Failed"
            $isSuccessful = $false
        }

        $row = [PSCustomObject]@{
            RowNumber = $rowIndex
            SourceSiteUrl = $siteUrl
            TargetSiteUrl = $targetSiteUrl
            SourceDocumentSetPath = $sourceDocSetPath
            TargetDocumentSetPath = $targetDocSetPath
            SourceItemCount = if ($null -ne $actionResult) { $actionResult.SourceItemCount } else { "" }
            TargetItemCount = if ($null -ne $actionResult) { $actionResult.TargetItemCount } else { "" }
            MovedItemCount = if ($null -ne $actionResult) { $actionResult.MovedItemCount } else { "" }
            RetainedItemCount = if ($null -ne $actionResult) { $actionResult.RetainedItemCount } else { "" }
            Status = $status
            IsSuccessful = $isSuccessful
            Message = $message
            Timestamp = (Get-Date).ToString("s")
        }

        Write-ResultRow -Row $row
        $script:ProcessedRows++

        if ($isSuccessful) {
            $script:SuccessfulRows++
            Write-Verbose "Row $rowIndex ($sourceDocSetPath): $message"
        }
        else {
            $script:FailedRows++
            Write-Warning "Row $rowIndex ($sourceDocSetPath): $message"
        }
    }

    $endTime = Get-Date
    $duration = $endTime - $startTime

    $summaryFile = Join-Path $OutputFolder "Summary_$(Get-Date -Format 'yyyyMMdd-HHmmss').json"
    $summary = @{
        ExecutionTime  = "$($duration.Hours)h $($duration.Minutes)m $($duration.Seconds)s"
        TotalRows      = $totalRows
        ProcessedRows  = $script:ProcessedRows
        SuccessfulRows = $script:SuccessfulRows
        FailedRows     = $script:FailedRows
        ResultFile     = $script:ResultFilePath
        PreviewFile    = if ($PreviewOnly) { $script:PreviewFilePath } else { "" }
        TimestampStart = $startTime.ToString("s")
        TimestampEnd   = $endTime.ToString("s")
    }

    $summary | ConvertTo-Json | Set-Content -Path $summaryFile -Encoding UTF8

    Write-Host ""
    Write-Host "==========================================" -ForegroundColor Cyan
    Write-Host "Document Set Split Summary" -ForegroundColor Cyan
    Write-Host "==========================================" -ForegroundColor Cyan
    Write-Host "Total Rows:       $($summary.TotalRows)"
    Write-Host "Processed Rows:   $($summary.ProcessedRows)"
    Write-Host "Successful Rows:  $($summary.SuccessfulRows)" -ForegroundColor Green
    Write-Host "Failed Rows:      $($summary.FailedRows)" -ForegroundColor $(if ($summary.FailedRows -gt 0) { "Red" } else { "Green" })
    Write-Host "Result CSV:       $($script:ResultFilePath)"
    if ($PreviewOnly) {
        Write-Host "Preview CSV:      $($script:PreviewFilePath)"
    }
    Write-Host "Summary JSON:     $summaryFile"
    Write-Host "Execution Time:   $($summary.ExecutionTime)"
    Write-Host "==========================================" -ForegroundColor Cyan
}

# ==========================================
# Main Execution
# ==========================================
try {
    Write-Host "Starting document set split cleanup..." -ForegroundColor Cyan
    Initialize-Credential
    Initialize-ResultWriter
    Invoke-SplitDocumentSetByItemCount
}
catch {
    Write-Error "Fatal error: $_"
    exit 1
}
finally {
    if ($script:ResultWriter) {
        try { $script:ResultWriter.Flush() } catch { }
        try { $script:ResultWriter.Dispose() } catch { }
    }
    if ($script:PreviewWriter) {
        try { $script:PreviewWriter.Flush() } catch { }
        try { $script:PreviewWriter.Dispose() } catch { }
    }
    Disconnect-PnPOnline -ErrorAction SilentlyContinue
    Write-Host "Script execution completed." -ForegroundColor Cyan
}
